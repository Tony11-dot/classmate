generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

enum Role {
  STUDENT
  TEACHER
  ADMIN
  PARENT
  SECRETARY
}

enum AccountStatus {
  AUTH_ONLY
  PENDING
  ACTIVE
}

enum ParentChildStatus {
  PENDING
  APPROVED
}

model User {
  id                      String                   @id @default(uuid())
  email                   String                   @unique
  password                String
  name                    String
  parentNotificationState ParentNotificationState?

  // NEW: identity/profile
  displayName String?
  legalName   String?
  phone       String?
  status      AccountStatus @default(AUTH_ONLY)

  // optional future tzfonet fields
  tzfonetEmail      String?
  tzfonetVerifiedAt DateTime?

  roles          UserRole[]
  studentProfile StudentProfile?

  // parent-child links
  parentLinks ParentChild[] @relation("ParentLinks")
  childLinks  ParentChild[] @relation("ChildLinks")

  // back-relation for teacher-owned courses
  taughtCourses Course[] @relation("TeacherCourses")

  createdAt             DateTime               @default(now())
  assessments           Assessment[]
  parentLinkCodes       ParentLinkCode[]
  announcements         Announcement[]
  announcementSeens     AnnouncementSeen[]
  parentNotifications   ParentNotification[]
  solutions             Solution[]
  notifications         Notification[]
  solutionLikes         SolutionLike[]
  solutionComments      SolutionComment[]
  teachingTemplateSlots ScheduleTemplateSlot[] @relation("TeacherSlots")

  studentTemplateBinds StudentScheduleTemplate[]
  solutionReposts      SolutionRepost[]
}

model UserRole {
  id     String @id @default(uuid())
  userId String
  role   Role
  user   User   @relation(fields: [userId], references: [id], onDelete: Cascade)

  @@unique([userId, role])
}

model Cohort {
  id    String @id @default(uuid())
  name  String @unique // e.g. "10th-2"
  grade Int // 7..12

  students              StudentProfile[]
  courses               Course[]
  scheduleSlots         ScheduleSlot[]
  joinCodes             CohortJoinCode[]
  scheduleOverrides     ScheduleOverride[]
  attendanceSessions    AttendanceSession[]
  scheduleTemplateBinds CohortScheduleTemplate[]
}

model CohortJoinCode {
  id        String    @id @default(uuid())
  cohortId  String
  codeHash  String
  active    Boolean   @default(true)
  createdAt DateTime  @default(now())
  expiresAt DateTime?

  cohort Cohort @relation(fields: [cohortId], references: [id], onDelete: Cascade)

  @@index([cohortId])
}

model StudentProfile {
  userId       String @id
  cohortId     String
  englishLevel Int
  mathLevel    Int

  user              User                 @relation(fields: [userId], references: [id], onDelete: Cascade)
  cohort            Cohort               @relation(fields: [cohortId], references: [id])
  studentMajors     StudentMajor[]
  enrollments       Enrollment[]
  attendanceRecords AttendanceRecord[]
  gradeRecords      GradeRecord[]
  // ===== AI BRAIN (v1) =====
  brainProfile      StudentBrainProfile?
  brainSkills       StudentBrainSkill[]
  brainEvents       StudentBrainEvent[]
  brainSummary      StudentBrainSummary?

  @@index([cohortId])
}

model ParentChild {
  id       String            @id @default(uuid())
  parentId String
  childId  String
  status   ParentChildStatus @default(APPROVED)

  parent User @relation("ParentLinks", fields: [parentId], references: [id], onDelete: Cascade)
  child  User @relation("ChildLinks", fields: [childId], references: [id], onDelete: Cascade)

  @@unique([parentId, childId])
  @@index([parentId])
  @@index([childId])
}

enum CourseRuleType {
  COHORT
  ENGLISH_LEVEL
  MATH_LEVEL
  MAJOR_REQUIRED
  ELECTIVE_OPT_IN
}

enum EnrollmentSource {
  AUTO
  MANUAL
}

model Major {
  id   String @id @default(uuid())
  name String @unique

  students StudentMajor[]
}

model StudentMajor {
  id        String @id @default(uuid())
  studentId String
  majorId   String

  student StudentProfile @relation(fields: [studentId], references: [userId], onDelete: Cascade)
  major   Major          @relation(fields: [majorId], references: [id], onDelete: Cascade)

  @@unique([studentId, majorId])
  @@index([majorId])
}

model Course {
  id        String  @id @default(uuid())
  name      String
  subject   String
  teacherId String?

  cohortId String?
  groupTag String?

  teacher User?   @relation("TeacherCourses", fields: [teacherId], references: [id])
  cohort  Cohort? @relation(fields: [cohortId], references: [id])

  rules              CourseRule[]
  slots              ScheduleSlot[]
  enrollments        Enrollment[]
  scheduleOverrides  ScheduleOverride[]
  attendanceSessions AttendanceSession[]
  assessments        Assessment[]
  templateSlots      ScheduleTemplateSlot[]

  messages    ClassroomMessage[]
  assignments ClassroomAssignment[]
  materials   ClassroomMaterial[]
  meetings    ClassroomMeeting[]

  @@index([teacherId])
  @@index([cohortId])
}

model CourseRule {
  id       String         @id @default(uuid())
  courseId String
  type     CourseRuleType
  value    String

  course Course @relation(fields: [courseId], references: [id], onDelete: Cascade)

  @@index([courseId])
}

model Enrollment {
  id        String           @id @default(uuid())
  courseId  String
  studentId String
  source    EnrollmentSource @default(AUTO)

  course  Course         @relation(fields: [courseId], references: [id], onDelete: Cascade)
  student StudentProfile @relation(fields: [studentId], references: [userId], onDelete: Cascade)

  @@unique([courseId, studentId])
  @@index([studentId])
}

model ScheduleSlot {
  id        String  @id @default(uuid())
  cohortId  String
  dayOfWeek Int
  period    Int
  courseId  String?

  cohort Cohort  @relation(fields: [cohortId], references: [id], onDelete: Cascade)
  course Course? @relation(fields: [courseId], references: [id])

  @@unique([cohortId, dayOfWeek, period])
  @@index([courseId])
}

model ScheduleOverride {
  id       String   @id @default(uuid())
  cohortId String
  date     DateTime // midnight UTC is fine; we'll treat as date in Asia/Jerusalem
  period   Int
  courseId String?

  cohort Cohort  @relation(fields: [cohortId], references: [id], onDelete: Cascade)
  course Course? @relation(fields: [courseId], references: [id])

  @@unique([cohortId, date, period])
  @@index([courseId])
  @@index([cohortId, date])
}

enum AttendanceStatus {
  PRESENT
  ABSENT
  LATE
